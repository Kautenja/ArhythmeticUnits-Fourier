\newcommand{\PanelScrew}[2]{%
  \draw[panel fine,fill=white] (#1,#2) circle[radius=5.2];
  \draw[panel fine] (#1-2.4,#2+2.4)--(#1+2.4,#2-2.4);
}
\newcommand{\PanelJack}[2]{%
  \begin{scope}[shift={(#1,#2)}]
    \draw[panel fine,fill=white] (0,0) circle[radius=11.5];
    \node[regular polygon,regular polygon sides=6,draw=panelInk,
      line width=.55pt,minimum size=20pt,inner sep=0pt] at (0,0) {};
    \draw[panel fine] (0,0) circle[radius=8.8];
    \draw[panel line,fill=white] (0,0) circle[radius=6.8];
    \draw[panel fine,fill=panelInk!12] (0,0) circle[radius=4.4];
  \end{scope}
}
\newcommand{\PanelKnob}[2]{%
